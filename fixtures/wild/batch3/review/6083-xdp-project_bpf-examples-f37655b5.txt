nat64: avoid 32-bit register spills when accessing skb->data

The verifier reports this error when accessing skb->data:

  ; switch (next_hdr_type) { @ parsing_helpers.h:149
  225: (bf) r3 = r10                    ; frame1: R3=fp0 R10=fp0
  226: (07) r3 += -56                   ; frame1: R3=fp-56
  227: (bf) r1 = r7                     ; frame1: R1=pkt(off=22,r=42) R7=pkt(off=22,r=42)
  228: (07) r1 += 12                    ; frame1: R1=pkt(off=34,r=42)
  ; void *data = (void *)(unsigned long long)skb->data; @ nat64_kern.c:70
  229: (61) r2 = *(u32 *)(r2 +76)       ; frame1: R2=pkt(r=0)
  ; csum = bpf_csum_diff((__be32 *)&iph->saddr, 2 * sizeof(__u32), @ nat64_kern.c:76
  230: (63) *(u32 *)(r10 -88) = r2
  invalid size of register spill

Apparently it's trying to spill only 32 bits from the register to the
stack, which is invalid. A similar problem was reported here:
https://github.com/cilium/cilium/pull/25336

Add some macros using inline asm to fix the problem. With this change
now the compiler properly generates 64-bit spills.

Signed-off-by: Beniamino Galvani <bgalvani@redhat.com>

+/* Macros to read the sk_buff data* pointers, preventing the compiler
+ * from generating a 32-bit register spill. */
+#define SKB_ACCESS_MEMBER_32(_skb, member)                       \
+({                                                               \
+	void *ptr;                                               \
+                                                                 \
+	asm volatile("%0 = *(u32 *)(%1 + %2)"                    \
+		     : "=r"(ptr)                                 \
+		     : "r"(_skb),                                \
+		       "i"(offsetof(struct __sk_buff, member))); \
+                                                                 \
+	ptr;                                                     \
+})
+
+#define SKB_DATA(_skb) SKB_ACCESS_MEMBER_32(_skb, data)
+#define SKB_DATA_END(_skb) SKB_ACCESS_MEMBER_32(_skb, data_end)
